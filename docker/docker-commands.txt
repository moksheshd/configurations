# MySQL
docker run \           
--name mysql \      
--publish 3306:3306 \
--env MYSQL_ROOT_PASSWORD=root \
--volume=~/data/mysql/data:/var/lib/mysql \
--detach \
mysql:8.0

# OpenHands Builder Application
export WORKSPACE_BASE="$HOME/workspace/personal/builder"
export SANDBOX_USER_ID=1000

docker run \
  --name openhands-app-builder \
  --pull=always \
  --interactive \
  --tty \
  --env SANDBOX_RUNTIME_CONTAINER_IMAGE=docker.all-hands.dev/all-hands-ai/runtime:0.15-nikolaik \
  --env LOG_ALL_EVENTS=true \
  --env SANDBOX_USER_ID=$SANDBOX_USER_ID \
  --env WORKSPACE_MOUNT_PATH=$WORKSPACE_BASE \
  --volume /var/run/docker.sock:/var/run/docker.sock \
  --volume $WORKSPACE_BASE:/opt/workspace_base \
  --publish 3000:3000 \
  --add-host host.docker.internal:host-gateway \
  --detach \
  docker.all-hands.dev/all-hands-ai/openhands:0.15
